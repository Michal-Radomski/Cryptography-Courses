You can implement AES encryption and decryption in Node.js using the built-in, cryptographically secure `crypto` module. It
is recommended to use `crypto.createCipheriv()` and `crypto.createDecipheriv()` with an authenticated mode like
**AES-256-GCM** (Galois/Counter Mode), which provides both confidentiality and data integrity.

Here is a complete example of how to implement AES-256-GCM encryption and decryption in Node.js:

```javascript
const crypto = require("crypto");

// Configuration constants
const ALGORITHM = "aes-256-gcm";
const KEY_LENGTH = 32; // 256 bits
const IV_LENGTH = 16; // 96 bits recommended for GCM

/**
 * Encrypts cleartext using AES-256-GCM
 * @param {string} text - The text to encrypt
 * @param {Buffer} key - The 32-byte secret key
 * @returns {Object} - Object containing hex-encoded encrypted data, iv, and authTag
 */
function encrypt(text, key) {
  // Generate a cryptographically random initialization vector
  const iv = crypto.randomBytes(IV_LENGTH);

  const cipher = crypto.createCipheriv(ALGORITHM, key, iv);

  let encrypted = cipher.update(text, "utf8", "hex");
  encrypted += cipher.final("hex");

  // Retrieve the authentication tag (crucial for GCM mode integrity checks)
  const authTag = cipher.getAuthTag().toString("hex");

  return {
    encryptedData: encrypted,
    iv: iv.toString("hex"),
    authTag: authTag,
  };
}

/**
 * Decrypts ciphertext using AES-256-GCM
 * @param {Object} encryptedObj - Object containing encryptedData, iv, and authTag
 * @param {Buffer} key - The 32-byte secret key
 * @returns {string} - The decrypted plain text
 */
function decrypt(encryptedObj, key) {
  const decipher = crypto.createDecipheriv(ALGORITHM, key, Buffer.from(encryptedObj.iv, "hex"));

  decipher.setAuthTag(Buffer.from(encryptedObj.authTag, "hex"));

  let decrypted = decipher.update(encryptedObj.encryptedData, "hex", "utf8");
  decrypted += decipher.final("utf8");

  return decrypted;
}

// --- Example Usage ---

// 1. Generate a secure random key (In a real application, securely load this from environment variables or a secrets manager)
const secretKey = crypto.randomBytes(KEY_LENGTH);

const sampleText = "This is a secret message that needs AES protection!";
console.log("Original Text:", sampleText);

// 2. Encrypt the text
const encryptedPackage = encrypt(sampleText, secretKey);
console.log("\nEncrypted Package:", encryptedPackage);

// 3. Decrypt the text back
const decryptedText = decrypt(encryptedPackage, secretKey);
console.log("\nDecrypted Text:", decryptedText);
```

### Key Security Best Practices:

- **Never hardcode keys:** Store your secret keys securely using environment variables (`process.env`) or a dedicated secret
  management service, rather than directly inside your source code.
- **Unique IVs:** Always use a unique initialization vector (`iv`) via `crypto.randomBytes(16)` for every encryption
  operation. Reusing an IV with the same key can completely compromise AES-GCM security.
- **Authentication Tags:** When using GCM mode, always verify the authentication tag (`setAuthTag` during decryption). If
  data is tampered with, the decryption function will automatically throw an error.
